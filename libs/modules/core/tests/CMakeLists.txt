set(tests config_registry_modules)

foreach(test ${tests})
  set(sources ${test}.cpp)

  source_group("Source Files" FILES ${sources})

  add_hpx_executable(
    ${test}_test INTERNAL_FLAGS
    SOURCES ${sources} ${${test}_FLAGS}
    EXCLUDE_FROM_ALL
    HPX_PREFIX ${HPX_BUILD_PREFIX}
    FOLDER "Tests/Unit/Modules/"
  )
  target_link_libraries(${test}_test PRIVATE hpx_core_module)

  add_hpx_unit_test("modules.cpp_modules" ${test} ${${test}_PARAMETERS})
endforeach()
